# Feudalize Domain Holdings — settled rulers only; tribal/nomad/herder → castle
sb_can_feudalize_government = {
	NOR = {
		government_has_flag = government_is_tribal
		government_has_flag = government_is_nomadic
	}
}

sb_province_is_feudalizable = {
	has_holding = yes
	has_ongoing_construction = no
	OR = {
		has_holding_type = tribal_holding
		has_holding_type = nomad_holding
		has_holding_type = herder_holding
	}
}

sb_domain_has_feudalizable_holding = {
	any_directly_owned_province = {
		sb_province_is_feudalizable = yes
	}
}

sb_can_afford_feudalize_plan = {
	gold >= var:sb_plan_cost
}
